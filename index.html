<script>
const ctx = new (window.AudioContext || window.webkitAudioContext)();
let prevOp = 1;
const kInf = 0.001515;
function playOp(n) {
if (ctx.state === 'suspended') ctx.resume();

const now = ctx.currentTime;
const oscV = ctx.createOscillator(); // Vector (Masa)
const oscR = ctx.createOscillator(); // Espinor R (Luz)
const gain = ctx.createGain();
// Frecuencia de Connes con No Conmutatividad
let freq = 137.036 * (1 + Math.log(n));
let mod = 1.0 + (n - prevOp) / 28.0;
freq *= mod;
prevOp = n;
// Simulación de Arrastre Geométrico S8
oscV.frequency.setValueAtTime(freq, now);
oscV.frequency.exponentialRampToValueAtTime(freq * (1 - kInf), now + 0.8);

oscR.frequency.setValueAtTime(freq * 1.5, now);
oscR.type = 'triangle';
gain.gain.setValueAtTime(0.4, now);
gain.gain.exponentialRampToValueAtTime(0.001, now + 0.8);
oscV.connect(gain); oscR.connect(gain);
gain.connect(ctx.destination);
oscV.start(); oscR.start();
oscV.stop(now + 0.8); oscR.stop(now + 0.8);
}
const board = document.getElementById('board');
for(let i=1; i<=28; i++) {
const el = document.createElement('div');
el.className = 'key';
el.innerHTML = i + '<span>OP</span>';
// Colores por Trialidad
const tri = i % 3;
el.style.borderColor = tri === 0 ? '#ff0055' : (tri === 1 ? '#00ffcc' : '#ffcc00');
el.ontouchstart = (e) => { e.preventDefault(); playOp(i); };
board.appendChild(el);
}
</script>
